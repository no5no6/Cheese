# 组件通信方式
+ ### `props`
  - 子组件 `child.vue`
  ```html
    <template>
      <div class="child-background">
        <h2>子组件</h2>
        <div>名称：{{ name }}</div>
        <div>分数：{{ count }}</div>
        <button @click="add">增加分数</button>
      </div>
    </template>

    <script setup lang="ts">
      // props 传参
      const props = defineProps(['name', 'count'])  
      // 自定义事件
      const $emit = defineEmits(['add'])  

      const add = () => {
        $emit('add', props.count + 1)
      }
    </script>

    <style scoped>
      .child-background {
        background-color: chartreuse;
        height: 200px;
        width: 200px;
      }
    </style>
  ``` 
  - 父组件 `App.vue`
  ```html
    <script setup lang="ts">
      import { ref } from 'vue'
      import Child from './components/child.vue' 

      const name = ref('张三')
      const count = ref(10)

      const handleAdd = (value: number) => {
        count.value = value
      }
    </script>

    <template>
      <div class="background">
        <h2>父组件</h2>
        <hr/>
        <Child :name="name" :count="count" @add="handleAdd"></Child>
      </div>
    </template>

    <style scoped>
      .background {
        display: flex;
        flex-direction: column;
        align-items: center;
        background-color: aqua;
        height: 400px;
        width: 400px;
      }
    </style>
  ```
