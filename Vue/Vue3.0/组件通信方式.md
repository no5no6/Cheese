# 组件通信方式
+ ### `props`
  - 子组件 `child.vue`
  ```html
    <template>
      <div class="child-background">
        <h2>子组件</h2>
        <div>名称：{{ name }}</div>
        <div>分数：{{ count }}</div>
        <button @click="add">增加分数</button>
      </div>
    </template>

    <script setup lang="ts">
      // props 传参
      const props = defineProps(['name', 'count'])  
      // 自定义事件
      const $emit = defineEmits(['add'])  

      const add = () => {
        $emit('add', props.count + 1)
      }
    </script>

    <style scoped>
      .child-background {
        background-color: chartreuse;
        height: 200px;
        width: 200px;
      }
    </style>
  ``` 
  - 父组件 `App.vue`
  ```html
    <script setup lang="ts">
      import { ref } from 'vue'
      import Child from './components/child.vue' 

      const name = ref('张三')
      const count = ref(10)

      const handleAdd = (value: number) => {
        count.value = value
      }
    </script>

    <template>
      <div class="background">
        <h2>父组件</h2>
        <hr/>
        <Child :name="name" :count="count" @add="handleAdd"></Child>
      </div>
    </template>

    <style scoped>
      .background {
        display: flex;
        flex-direction: column;
        align-items: center;
        background-color: aqua;
        height: 400px;
        width: 400px;
      }
    </style>
  ```

+ ### `event bus` (通过 `mitt` 插件 实现)
  - 安装 `mitt` 插件
    `pnpm add mitt --save`
  - 添加 `src/utils/index.ts` 文件  

    ```typescript
      import mitt from 'mitt'

      type Events = {
        add: number // 要使用的 方法（add）: 参数类型：（number）
      };

      const emitter= mitt<Events>()

      export default emitter
    ```
  - App.vue
    ```html
      <script setup lang="ts">
        import Child from './components/child.vue'
        import Child2 from './components/child2.vue'

      </script>

      <template>
        <div class="background">
          <h2>父组件</h2>
          <div class="child">
            <Child></Child>
            <Child2></Child2>
          </div>
        </div>
      </template>

      <style scoped>
      .background {
        background-color: aqua;
        height: 400px;
        width: 800px;
      }

      .child {
        height: 400px;
        display: flex;
        justify-content: space-between;
        align-items: center;
      }
      </style>
    ```
  - 添加 `src/components/child.vue` 文件
    ```html
      <template>
        <div class="child-background">
          <h2>第一个子组件</h2>
          <div>余额：{{ money }}</div>
        </div>
      </template>

      <script setup lang="ts">

      import emitter from '../utils/index'
      import { ref, onMounted } from 'vue'

      const money = ref(0)

      onMounted(() => {
        emitter.on('add', count => {
          money.value += count
        })
      })

      </script>

      <style scoped>
      .child-background {
        background-color: chartreuse;
        height: 200px;
        width: 200px;
      }
      </style>
    ```
  - 添加 `src/components/child2.vue` 文件
    ```html
      <template>
        <div class="child-background">
          <h2>第二个子组件</h2>
          <button @click="add">转账</button>
        </div>
      </template>

      <script lang="ts" setup>
      import emitter from '../utils/index'

      const add = ()=> {
        emitter.emit('add', 1)
      }
      </script>

      <style scoped>
        .child-background {
          background-color: blueviolet;
          height: 200px;
          width: 200px;
        }
      </style>
    ```
